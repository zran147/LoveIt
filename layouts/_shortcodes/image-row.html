{{- /* image-row: place several `image` shortcodes side by side in one row. Usage: {{< image-row height=320 >}} {{< image src="a.png" caption="a" >}} {{< image src="b.png" caption="b" >}} {{< /image-row >}} Params: height [optional] target height in px for the tallest image (e.g. 320). All images are scaled by the same factor so the tallest one is exactly this tall; aspect ratios are preserved. mode [optional] "fit" (default, scale to `height` as described above) "equal" (equal-width columns, cropped to height) gap [optional] space between images, default "4rem" class [optional] extra class on the wrapper */ -}} {{- $height := .Get "height" | default 0 | int -}} {{- $mode := .Get "mode" | default "fit" -}} {{- $gap := .Get "gap" | default "4rem" -}} {{- $class := .Get "class" | default "" -}} {{- $h := "none" -}} {{- $scale := 1.0 -}} {{- if gt $height 0 -}} {{- $h = printf "%dpx" $height -}} {{- /* collect each image's natural height (emitted as `--ih` by the img partial) and derive a common scale so the tallest == $height */ -}} {{- $heights := slice -}} {{- range findRE `--ih:\s*[0-9]+px` .Inner -}} {{- $heights = $heights | append (replaceRE `[^0-9]` "" . | int) -}} {{- end -}} {{- $max := 0 -}} {{- range $heights -}} {{- if gt . $max -}}{{- $max = . -}}{{- end -}} {{- end -}} {{- if gt $max 0 -}} {{- $scale = div (float $height) (float $max) -}} {{- end -}} {{- end -}}
