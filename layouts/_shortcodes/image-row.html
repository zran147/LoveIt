{{- /* image-row: place several `image` shortcodes side by side in one row. Usage: {{< image-row height=320 >}} {{< image src="a.png" caption="a" >}} {{< image src="b.png" caption="b" >}} {{< /image-row >}} Params: height [optional] target height in px for the tallest image (e.g. 320). All images are scaled by the same factor so the tallest one is exactly this tall; aspect ratios are preserved. mode [optional] "fit" (default, scale to `height` as described above) "equal" (equal-width columns, cropped to height) gap [optional] space between images, default "4rem" class [optional] extra class on the wrapper Alignment: if the row height is taller than half the viewport the images are top-aligned, otherwise they are middle-aligned (pure CSS, per-row media query, reacts to resizing). */ -}} {{- $height := .Get "height" | default 0 | int -}} {{- $mode := .Get "mode" | default "fit" -}} {{- $gap := .Get "gap" | default "4rem" -}} {{- $class := .Get "class" | default "" -}} {{- $h := "none" -}} {{- $scale := 1.0 -}} {{- /* collect each image's natural height (emitted as `--ih` by the img partial) and derive a common scale so the tallest == $height */ -}} {{- $heights := slice -}} {{- range findRE `--ih:\s*[0-9]+px` .Inner -}} {{- $heights = $heights | append (replaceRE `[^0-9]` "" . | int) -}} {{- end -}} {{- $maxH := 0 -}} {{- range $heights -}} {{- if gt . $maxH -}}{{- $maxH = . -}}{{- end -}} {{- end -}} {{- if gt $height 0 -}} {{- $h = printf "%dpx" $height -}} {{- if gt $maxH 0 -}} {{- $scale = div (float $height) (float $maxH) -}} {{- end -}} {{- end -}} {{- /* Height the row will take up: the requested `height`, or the tallest rendered image height (`--rh`, i.e. taking any per-image width/height params into account) when no `height` is given. If that is taller than half the viewport the row is top-aligned, otherwise the images are middle-aligned. Done with a per-row media query (no JS, reacts to resize). */ -}} {{- $rowH := $height -}} {{- if le $rowH 0 -}} {{- $rhs := slice -}} {{- range findRE `--rh:\s*[0-9]+px` .Inner -}} {{- $rhs = $rhs | append (replaceRE `[^0-9]` "" . | int) -}} {{- end -}} {{- range $rhs -}} {{- if gt . $rowH -}}{{- $rowH = . -}}{{- end -}} {{- end -}} {{- if le $rowH 0 -}}{{- $rowH = $maxH -}}{{- end -}} {{- end -}} {{- $id := printf "img-row-%d" .Ordinal -}}
{{- .Inner -}}
{{- if gt $rowH 0 -}} {{- end -}}
