{{- /* lazysizes and lightgallery */ -}} {{- $src := .Src -}} {{- $userW := .Width -}} {{- $userH := .Height -}} {{- $natW := 0 -}} {{- $natH := 0 -}} {{- with dict "Path" $src "Resources" .Resources | partial "function/resource.html" -}} {{- $src = .RelPermalink -}} {{ if not (eq .MediaType.SubType "svg") }} {{- $natW = .Width -}} {{- $natH = .Height -}} {{ end }} {{- end -}} {{- $small := .SrcSmall | default $src -}} {{- with dict "Path" .SrcSmall "Resources" .Resources | partial "function/resource.html" -}} {{- $small = .RelPermalink -}} {{- end -}} {{- $large := .SrcLarge | default $src -}} {{- with dict "Path" .SrcLarge "Resources" .Resources | partial "function/resource.html" -}} {{- $large = .RelPermalink -}} {{- end -}} {{- $alt := .Alt | default $src -}} {{- $loading := resources.Get "svg/loading.svg" | minify -}} {{- /* Inline style with (a) the natural dimensions as CSS vars — used by the `image-row` shortcode — and (b) an explicit size requested through the `image` shortcode's `width` / `height` params. A `height` is converted to the equivalent width so the aspect ratio is preserved and the column width still caps it (no distortion, no overflow). */ -}} {{- $style := "" -}} {{- with $natW }}{{ $style = printf "--iw: %vpx" . }}{{ end -}} {{- with $natH }}{{ $style = printf "%s; --ih: %vpx" $style . }}{{ end -}} {{- /* effective (rendered) height — used by `image-row` to decide alignment */ -}} {{- $rh := $natH -}} {{- if $userH -}} {{- $rh = $userH | replaceRE "[^0-9]" "" | int -}} {{- else if $userW -}} {{- if and (gt $natW 0) (gt $natH 0) -}} {{- $rh = int (div (mul ($userW | replaceRE "[^0-9]" "" | int) $natH) $natW) -}} {{- end -}} {{- end -}} {{- if gt $rh 0 }}{{ $style = printf "%s; --rh: %dpx" $style $rh }}{{ end -}} {{- if $userW -}} {{- with ($userW | replaceRE "[^0-9]" "") -}} {{- $style = printf "%s; width: %vpx; height: auto" $style (int .) -}} {{- end -}} {{- else if $userH -}} {{- $h := $userH | replaceRE "[^0-9]" "" | int -}} {{- if and (gt $h 0) (gt $natH 0) -}} {{- $style = printf "%s; width: %dpx; height: auto" $style (int (div (mul $h $natW) $natH)) -}} {{- else if gt $h 0 -}} {{- $style = printf "%s; height: %dpx; width: auto" $style $h -}} {{- end -}} {{- end -}} {{- if .Linked -}} {{- else -}} {{- end -}}
